-- Helper function to generate unique pet name
local function GenerateUniquePetName()
    local prefixes = {"★", "☆", "♡", "☁️", "✨", "🍓", "🌸", "🍯", "☕", "🌙", "🌈", "❄️", "🫧", "🍬", "🍪", "🥛"}
    local names = {"Shadow", "Blaze", "Frost", "Thunder", "Moon", "Star", "Sky", "Ocean", "River", "Storm",   
                   "Ember", "Ash", "Dusk", "Dawn", "Night", "Day", "Sun", "Wind", "Rain", "Snow", "Ice", "Fire",
                   "Nova", "Cosmo", "Galaxy", "Orbit", "Comet", "Meteor", "Aurora", "Nebula", "Crystal", "Gem",
                   "Ruby", "Sapphire", "Emerald", "Diamond", "Gold", "Silver", "Mystic", "Magic", "Enchant"}
    
    local usePrefix = math.random(1, 3) == 1
    local name = names[math.random(1, #names)]
    
    if usePrefix then
        return prefixes[math.random(1, #prefixes)] .. name
    else
        return name .. " " .. prefixes[math.random(1, #prefixes)]
    end
end

-- Helper function to get property group
local function GetPropertyGroup(properties)
    local isMega = properties.mega_neon or false
    local isNeon = properties.neon or false
    local canFly = properties.flyable or false
    local canRide = properties.rideable or false

    if isMega then
        if canFly and canRide then return "mega_neon_flyable_rideable"
        elseif canFly then return "mega_neon_flyable"
        elseif canRide then return "mega_neon_rideable"
        else return "mega_neon" end
    elseif isNeon then
        if canFly and canRide then return "neon_flyable_rideable"
        elseif canFly then return "neon_flyable"
        elseif canRide then return "neon_rideable"
        else return "neon" end
    else
        if canFly and canRide then return "flyable_rideable"
        elseif canFly then return "flyable"
        elseif canRide then return "rideable"
        else return "regular" end
    end
end

    }, {

-- Apply neon effect to a pet model
local function applyNeonToModel(petModel, petData, isMega)
    if not petModel then return end
    if not PetRigs then return end  -- add this
    local pm = petModel:FindFirstChild("PetModel") or petModel
    local kindData = KindDB[petData.id]
    if not kindData or not kindData.neon_parts then 
        print("No neon parts for this pet")
        return 
    end
    
    print("Applying neon to:", petData.id, "Mega:", isMega or false)
    
    -- Apply neon to all neon parts
    for partName, config in pairs(kindData.neon_parts) do
        local part = PetRigs.get(pm).get_geo_part(pm, partName)
        if part then
            part.Material = Enum.Material.Neon
            if isMega then
                -- Mega neon - brighter colors
                local h, s, v = config.Color:ToHSV()
                part.Color = Color3.fromHSV(h, math.min(s * 1.3, 1), math.min(v * 1.4, 1))
            else
                -- Regular neon - original colors
                part.Color = config.Color
            end
        end
    end
end

-- Create item (pet or toy) - from document
local function createItem(itemId, category, properties)
    local uniqueId = genUID()
    local itemKindData = KindDB[itemId]

    if not itemKindData then
        warn("Item ID not found: " .. itemId)
        return nil
    end

    properties = properties or {}
    local newnessOrder = NextToyOrder

    -- Use property grouping only for pets
    if category == "pets" then
        local groupKey = GetPropertyGroup(properties)
        NewnessGroups[groupKey] = NewnessGroups[groupKey] - 1
        newnessOrder = NewnessGroups[groupKey]

        -- Set ailments_completed to 0 to prevent warning (but no actual ailments)
        if not properties.ailments_completed then
            properties.ailments_completed = 0
        end

        -- Generate unique name if not provided
        if not properties.rp_name or properties.rp_name == "" then
            properties.rp_name = GenerateUniquePetName()
        end

        -- Add friendship level for mega pets
        if properties.mega_neon and not properties.friendship_level then
            properties.friendship_level = math.random(1, 5)
        end
    else
        NextToyOrder = NextToyOrder - 1
        newnessOrder = NextToyOrder
    end

    local itemData = {
        unique = uniqueId,
        category = category,
        id = itemId,
        kind = itemKindData.kind,
        newness_order = newnessOrder,
        properties = properties,
        _source = "joeybala"  -- Mark as spawned by our script
    }

    local identity = get_thread_identity and get_thread_identity() or 8
    set_thread_identity(2)
    local inventory = ClientData.get("inventory")
    if inventory and inventory[category] then
        inventory[category][uniqueId] = itemData
    end
    set_thread_identity(identity)

    -- Track in activePets only if it's a pet
    if category == "pets" then
        activePets[uniqueId] = {
            data = itemData,
            model = nil
        }
    end

    SpawnedItems[uniqueId] = true

    task.defer(function()
        if UIManager and UIManager.apps and UIManager.apps.BackpackApp then
            UIManager.apps.BackpackApp:refresh_rendered_items()
        end
    end)

    return itemData
end

-- Add/remove wrappers & state managers
local function addCharWrapper(petModel, petData)
    updateData("pet_char_wrappers", function(w)
        table.insert(w, {
            char = petModel,
            mega_neon = petData.properties.mega_neon or false,
            neon = petData.properties.neon or false,
            player = LocalPlayer,
            entity_controller = LocalPlayer,
            controller = LocalPlayer,
            rp_name = petData.properties.rp_name or "",
            pet_trick_level = petData.properties.pet_trick_level or 0,
            pet_unique = petData.unique,
            pet_id = petData.id,
            location = { full_destination_id = "housing", destination_id = "housing", house_owner = LocalPlayer },
            pet_progression = { age = petData.properties.age or 1, percentage = math.random() },
            are_colors_sealed = false,
            is_pet = true,
        })
        for idx, wrap in ipairs(w) do wrap.unique = idx; wrap.index = idx end
        return w
    end)
end
local function addStateManager(petModel)
    updateData("pet_state_managers", function(s)
        table.insert(s, { char = petModel, player = LocalPlayer, store_key = "pet_state_managers", is_sitting = false, chars_connected_to_me = {}, states = {} })
        return s
    end)
end
local function removeCharWrapper(petUnique)
    updateData("pet_char_wrappers", function(w)
        local idx = findIndex(w, function(v) return v.pet_unique == petUnique end)
        if idx then table.remove(w, idx) end
        for i, v in ipairs(w) do v.unique = i; v.index = i end
        return w
    end)
end
local function removeStateManager(petModel)
    updateData("pet_state_managers", function(s)
        local idx = findIndex(s, function(v) return v.char == petModel end)
        if idx then table.remove(s, idx) end
        return s
    end)
end
local function clearPetState(petModel)
    updateData("pet_state_managers", function(s)
        local idx = findIndex(s, function(v) return v.char == petModel end)
        if idx then
            local new = table.clone(s)
            new[idx] = table.clone(new[idx])
            new[idx].states = {}
            return new
        end
        return s
    end)
end
local function setPetState(petModel, stateId)
    updateData("pet_state_managers", function(s)
        local idx = findIndex(s, function(v) return v.char == petModel end)
        if idx then
            local new = table.clone(s)
            new[idx] = table.clone(new[idx])
            new[idx].states = { { id = stateId } }
            return new
        end
        return s
    end)
end
local function clearPlayerState()
    updateData("state_manager", function(st)
        local c = table.clone(st)
        c.states = {}; c.is_sitting = false
        return c
    end)
end
local function setPlayerState(stateId)
    updateData("state_manager", function(st)
        local c = table.clone(st)
        c.states = { { id = stateId } }; c.is_sitting = true
        return c
    end)

local function attachToPet(petModel)
    local char = LocalPlayer.Character
    if not char or not char.PrimaryPart then return false end
    local ridePos = petModel:FindFirstChild("RidePosition", true)
    if not ridePos then return false end
    local att = Instance.new("Attachment")
    att.Parent = ridePos
    att.Position = Vector3.new(0,1.237,0)
    att.Name = "SourceAttachment"
    local constraint = Instance.new("RigidConstraint")
    constraint.Name = "StateConnection"
    constraint.Attachment0 = att
    constraint.Attachment1 = char.PrimaryPart.RootAttachment
    constraint.Parent = char
    return true
end
local function mount(petUnique, playerState, petState)
    local pet = activePets[petUnique]
    if not pet or not pet.model then return end
    local char = LocalPlayer.Character
    if not char or not char.Humanoid then return end
    mountedPet = petUnique
    setPetState(pet.model, petState)
    setPlayerState(playerState)
    pet.model:ScaleTo(2)
    attachToPet(pet.model)
    if mountTrack then mountTrack:Stop(); mountTrack:Destroy() end
    mountTrack = char.Humanoid.Animator:LoadAnimation(AnimationManager.get_track("PlayerRidingPet"))
    char.Humanoid.Sit = true
    for _, part in pairs(char:GetDescendants()) do
        if part:IsA("BasePart") and not part.Massless then
            part.Massless = true
            part:SetAttribute("HaveMass", true)
        end
    end
    mountTrack:Play()
end
local function unmount()
    if not mountedPet then return end
    local pet = activePets[mountedPet]
    if pet and pet.model then
        if mountTrack then mountTrack:Stop(); mountTrack:Destroy(); mountTrack = nil end
        local att = pet.model:FindFirstChild("SourceAttachment", true)
        if att then att:Destroy() end
        local char = LocalPlayer.Character
        if char then
            for _, part in pairs(char:GetDescendants()) do
                if part:IsA("BasePart") and part:GetAttribute("HaveMass") then
                    part.Massless = false
                end
            end
        end
        clearPetState(pet.model)
        clearPlayerState()
        pet.model:ScaleTo(1)
    end
    mountedPet = nil
end
local function ridePet(unique) mount(unique, "PlayerRidingPet", "PetBeingRidden") end
local function flyPet(unique) mount(unique, "PlayerFlyingPet", "PetBeingFlown") end

-- Equip/Unequip
local function unequip(petData)
    local pet = activePets[petData.unique]
    if not pet or not pet.model then return end
    if mountedPet == petData.unique then unmount() end
    removeCharWrapper(petData.unique)
    removeStateManager(pet.model)
    pet.model:Destroy()
    pet.model = nil
    if equippedPet and equippedPet.unique == petData.unique then equippedPet = nil end
end
local function equip(petData)
    if petData.category ~= "pets" then return end
    if equippedPet then unequip(equippedPet) end
    -- unequip any real pets
    for _, w in pairs(ClientData.get("pet_char_wrappers") or {}) do
        if w.controller == LocalPlayer then
            pcall(function() OriginalRouterGet("ToolAPI/Unequip"):InvokeServer(w.pet_unique) end)
        end
    end
    if not activePets[petData.unique] then
        activePets[petData.unique] = { data = petData, model = nil }
    end
    local model = getPetModel(petData.kind):Clone()
    model.Parent = workspace
    activePets[petData.unique].model = model
    
    -- Apply neon/mega neon effect if properties exist
    if petData.properties.neon or petData.properties.mega_neon then
        applyNeonToModel(model, petData, petData.properties.mega_neon or false)
    end
    
    equippedPet = petData
    
    task.defer(function()
        addCharWrapper(model, petData)
        addStateManager(model)
        
        -- Play fusion effect for neon/mega neon pets
        if petData.properties.neon or petData.properties.mega_neon then
            task.wait(0.2)
            local wrapper = nil
            for _, w in pairs(ClientData.get("pet_char_wrappers") or {}) do
                if w.pet_unique == petData.unique then
                    wrapper = w
                    break
                end
            end
            if wrapper then
                playFusionEffect(wrapper, petData.properties.mega_neon or false)
            end
        end
    end)
end

-- Function to perform a trick on equipped pet
local function doPetTrick(trickName)
    -- Get the equipped pet entity
    local entities = PetEntityManager.get_local_owned_pet_entities()
    if not entities or #entities == 0 then
        warn("❌ No pet equipped!")
        return false
    end
    
    -- Get the most recent pet
    local entity = entities[#entities]
    local petModel = entity.base.pet_model
    if not petModel then
        warn("❌ Pet model not found!")
        return false
    end
    
    -- Map common trick names to animation IDs
    local trickAnimations = {
        ["Sit"] = "Sit",
        ["Lay Down"] = "LayDown",
        ["Joy"] = "Joy",
        ["Jump"] = "Jump",
        ["Twirl Around"] = "Twirl",
        ["Triple Flip"] = "TripleFlip"
    }
    
    local animName = trickAnimations[trickName]
    if not animName then
        animName = trickName -- Use as-is if not in map
    end
    
    -- Get the animation track
    local track = AnimationManager.get_track(animName)
    if not track then
        warn("❌ Animation track not found:", animName)
        return false
    end
    
    -- Get or create animator
    local animator = petModel:FindFirstChildWhichIsA("Animator", true)
    if not animator then
        local ac = petModel:FindFirstChildOfClass("AnimationController")
        if not ac then
            ac = Instance.new("AnimationController")
            ac.Parent = petModel
        end
        animator = ac:FindFirstChildOfClass("Animator")
        if not animator then
            animator = Instance.new("Animator")
            animator.Parent = ac
        end
    end
    
    -- Stop any currently playing animations
    for _, playingTrack in ipairs(animator:GetPlayingAnimationTracks()) do
        playingTrack:Stop()
    end
    
    -- Load and play the animation
    local success = pcall(function()
        local animTrack = animator:LoadAnimation(track)
        animTrack:Play()
        print("✅ Playing trick:", trickName)
    end)
    
    if not success then
        warn("❌ Failed to play trick:", trickName)
        return false
    end
    
    return true
end

-- Auto-capitalize helper function
local function autoCapitalize(text)
    if not text or text == "" then
        return text
    end
    
    local result = ""
    local capitalizeNext = true
    
    for i = 1, #text do
        local char = text:sub(i, i)
        if char == " " then
            capitalizeNext = true
            result = result .. char
        elseif capitalizeNext then
            result = result .. char:upper()
            capitalizeNext = false
        else
            result = result .. char:lower()
        end
    end
    
    return result
end

-- ================================================================
-- DRAG FUNCTIONALITY
-- ================================================================
local dragging = false
local dragStart = nil
local startPos = nil

mainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = mainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
                dragging = false
            end
        end)
    end
end)

mainFrame.InputChanged:Connect(function(input)
    if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local delta = input.Position - dragStart
        mainFrame.Position = UDim2.new(
            startPos.X.Scale,
            startPos.X.Offset + delta.X,
            startPos.Y.Scale,
            startPos.Y.Offset + delta.Y
        )
    end
end)

local function hexToColor3(hex)
    hex = hex:gsub("#",""):gsub("%s+","")
    if #hex ~= 6 then return nil end
    return Color3.fromRGB(tonumber(hex:sub(1,2),16), tonumber(hex:sub(3,4),16), tonumber(hex:sub(5,6),16))
end

        age = math.random(1, 6),

        rp_name = "",
        friendship_level = activeFlags["M"] and math.random(1, 5) or 1
    }
    
    local item = _G.spawn_pet(name, options)
    
    if item then
        if activeFlags["M"] or activeFlags["N"] then
            task.spawn(function()
                task.wait(0.5)
                local pet = activePets[item.unique]
                if pet and pet.model then
                    applyNeonToModel(pet.model, pet.data, activeFlags["M"])
                    task.wait(0.1)
                    for _, w in pairs(ClientData.get("pet_char_wrappers") or {}) do
                        if w.pet_unique == item.unique then
                            playFusionEffect(w, activeFlags["M"])
                            break
                        end
                    end
                end
            end)
        end
        spawnPetBtn.Text = "✓"
        task.wait(0.3)
        spawnPetBtn.Text = "Spawn Pet"
    end

    local name = wearBox.Text
    if name == "" then return end
    local id = getPetWearId(name)
    if id then
        local PetAccessoryEquipHelper = Load("PetAccessoryEquipHelper")
        local wearEntry = InventoryDB.pet_accessories and InventoryDB.pet_accessories[id]
        if wearEntry then
            local wrappers = ClientData.get("pet_char_wrappers")
            if wrappers and #wrappers > 0 then
                local petModel = wrappers[#wrappers].char
                if petModel then
                    local asset = DownloadClient.promise_download_copy("PetAvatarResources", wearEntry.model_handle):expect()
                    PetAccessoryEquipHelper.equip_accessory({
                        pet_model = petModel,
                        accessory_base_asset = asset,
                        accessory_item_entry = wearEntry,
                        play_poof_effect = true,
                        is_mannequin = false,
                    })
                    addWearBtn.Text = "✓"
                    task.wait(0.3)
                    addWearBtn.Text = "Add PetWear"
                end
            end
        end
    end
end)
