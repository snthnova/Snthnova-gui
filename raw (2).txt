local function isAlive(plr)

    local c = plr and plr.Character

    local h = c and c:FindFirstChildOfClass("Humanoid")

    return h and h.Health > 0

end

local function getRole(plr)

    local map = Workspace:FindFirstChild("Map")

    if not map then return "Innocent" end

    local roles = map:FindFirstChild("roles")

    if not roles then return "Innocent" end

    local murderers = roles:FindFirstChild("Murderers")

    if murderers then

        for _, v in ipairs(murderers:GetChildren()) do

            if v.Name == plr.Name then return "Murderer" end

        end

    end

    local sheriffs = roles:FindFirstChild("Sheriffs")

    if sheriffs then

        for _, v in ipairs(sheriffs:GetChildren()) do

            if v.Name == plr.Name then return "Sheriff" end

        end

    end

    return "Innocent"

end

local function worldToVP(pos)

    local vp, onScreen = Cam:WorldToViewportPoint(pos)

    return Vector2.new(vp.X, vp.Y), onScreen, vp.Z

end

local function getClosestPlayer(fov, ignoreWall)

    local bestDist = fov

    local bestPlr  = nil

    local myHRP    = getHRP()

    if not myHRP then return nil end

    local center   = Vector2.new(Cam.ViewportSize.X / 2, Cam.ViewportSize.Y / 2)

    for _, plr in ipairs(Players:GetPlayers()) do

        if plr ~= LP and isAlive(plr) then

            if cfg.AimbotTeamCheck then        

                local myRole  = getRole(LP)

                local tgtRole = getRole(plr)

                if myRole == tgtRole and myRole ~= "Innocent" then continue end

            end

            local char = plr.Character

            local bone = getBone(char, cfg.AimbotBone)

            if bone then

                local vp, onScreen = worldToVP(bone.Position)

                if onScreen then

                    local dist = (vp - center).Magnitude

                    if dist < bestDist then

                        if ignoreWall or not cfg.AimbotWallCheck then

                            bestDist = dist

                            bestPlr  = plr

                        else

                            local ray = Ray.new(myHRP.Position,

                                (bone.Position - myHRP.Position).Unit * 1000)

                            local hit = Workspace:FindPartOnRayWithIgnoreList(

                                ray, {getChar(), char})

                            if not hit then

                                bestDist = dist

                                bestPlr  = plr

                            end

                        end

                    end

                end

            end

        end

    end

    return bestPlr

end

local Predictions = {}

function Predictions.Ubuntu(hrp, dist)

    if not hrp then return Vector3.zero end

    local vel = hrp.Velocity

    local latency = dist / 500  

    return hrp.Position + vel * latency

end

function Predictions.Remi(hrp, dist)

    if not hrp then return Vector3.zero end

    local vel = hrp.Velocity

    local t   = dist / 500

    local gravity = Vector3.new(0, -Workspace.Gravity * 0.5 * t * t, 0)

    return hrp.Position + vel * t + gravity

end

function Predictions.Electro(hrp, dist)

    if not hrp then return Vector3.zero end

    local vel = hrp.Velocity

    local dt  = RunService.Heartbeat:Wait()

    local smoothed = vel * (dist / 400) * (1 - dt * 2)

    return hrp.Position + smoothed

end

function Predictions.Rectum(hrp, dist)

    if not hrp then return Vector3.zero end

    local vel = hrp.Velocity

    local t   = (dist / 350) * 1.25

    return hrp.Position + vel * t

end

function Predictions.Tiancai(hrp, dist)

    if not hrp then return Vector3.zero end

    local vel = hrp.Velocity

    local t   = dist / 450

    local lateral = Vector3.new(vel.X * t * 1.1, 0, vel.Z * t * 1.1)

    local vertical = Vector3.new(0, vel.Y * t - (Workspace.Gravity * 0.25 * t * t), 0)

    return hrp.Position + lateral + vertical

end

function Predictions.Revert(hrp, dist)

    if not hrp then return Vector3.zero end

    return hrp.Position

end

function Predictions.Dynamic(hrp, dist)

    if not hrp then return Vector3.zero end

    local vel = hrp.Velocity

    local t   = dist / 120

    return hrp.Position + vel * t

end

local function getPredictedPos(hrp, mode)

    local myHRP = getHRP()

    if not myHRP or not hrp then return hrp and hrp.Position or Vector3.zero end

    local dist = (hrp.Position - myHRP.Position).Magnitude

    local fn = Predictions[mode] or Predictions.Ubuntu

    return fn(hrp, dist)

end

local function initGunSilentAim()

    local remote = Workspace:FindFirstChild("SheriffGun_REMOTE", true)

    if not remote then return end

    local orig = remote.FireServer

    hookfunction(orig, function(self, ...)

        if not cfg.GunSilentAim then return orig(self, ...) end

        local args = {...}

        local target = getClosestPlayer(cfg.AimbotFOV, cfg.GunSilentAimWallCheck)

        if target and target.Character then

            local hrp = target.Character:FindFirstChild("HumanoidRootPart")

            if hrp then

                local predicted = getPredictedPos(hrp, cfg.GunSilentAimPrediction)

                args[1] = predicted 

                if cfg.GunSilentAimAlwaysHit then

                    args[2] = hrp  

                end

            end

        end

        return orig(self, table.unpack(args))

    end)

end

local flyBody = nil

local flyBodyGyro = nil

local function enableFly()

    local hrp = getHRP()

    if not hrp then return end

    local hum = getHumanoid()

    if hum then hum.PlatformStand = true end

    flyBody = Instance.new("BodyVelocity")

    flyBody.MaxForce   = Vector3.new(1e9, 1e9, 1e9)

    flyBody.Velocity   = Vector3.zero

    flyBody.Name       = "ODH_FlyVelocity"

    flyBody.Parent     = hrp

    flyBodyGyro = Instance.new("BodyGyro")

    flyBodyGyro.MaxTorque  = Vector3.new(1e9, 1e9, 1e9)

    flyBodyGyro.P          = 1e4

    flyBodyGyro.D          = 100

    flyBodyGyro.CFrame     = hrp.CFrame

    flyBodyGyro.Name       = "ODH_FlyGyro"

    flyBodyGyro.Parent     = hrp

end

local function disableFly()

    if flyBody    then flyBody:Destroy();    flyBody    = nil end

    if flyBodyGyro then flyBodyGyro:Destroy(); flyBodyGyro = nil end

    local hum = getHumanoid()

    if hum then hum.PlatformStand = false end

end

RunService.Heartbeat:Connect(function(dt)

    if not cfg.FlyEnabled then

        if flyBody then disableFly() end

        return

    end

    if not flyBody then enableFly() end

    if not flyBody then return end

    local speed = cfg.FlySpeed

    local dir   = Vector3.zero

    if UserInputService:IsKeyDown(Enum.KeyCode.W) then

        dir = dir + Cam.CFrame.LookVector end

    if UserInputService:IsKeyDown(Enum.KeyCode.S) then

        dir = dir - Cam.CFrame.LookVector end

    if UserInputService:IsKeyDown(Enum.KeyCode.A) then

        dir = dir - Cam.CFrame.RightVector end

    if UserInputService:IsKeyDown(Enum.KeyCode.D) then

        dir = dir + Cam.CFrame.RightVector end

    if UserInputService:IsKeyDown(Enum.KeyCode.Space) then

        dir = dir + Vector3.new(0, 1, 0) end

    if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then

        dir = dir - Vector3.new(0, 1, 0) end

    if dir.Magnitude > 0 then

        flyBody.Velocity = dir.Unit * speed

    else

        flyBody.Velocity = Vector3.zero

    end

    if flyBodyGyro then

        flyBodyGyro.CFrame = Cam.CFrame

    end

end)

local tradeHistory = {}   -- { {time, partner, yourItems, theirItems, verdict} }

local function hookTradeSystem()

    local tradeRemote = ReplicatedStorage:FindFirstChild("TradingSystemRemote", true)

    if not tradeRemote then return end

    local orig = tradeRemote.OnClientEvent

    tradeRemote.OnClientEvent:Connect(function(action, data)

        if not cfg.TradeTrackerEnabled then return end

        if action == "TradeComplete" or action == "TradeResult" then

            local entry = {

                time       = os.date("%H:%M:%S"),

                partner    = data and data.partner or "Unknown",

                yourItems  = data and data.yourItems or {},

                theirItems = data and data.theirItems or {},

                verdict    = data and data.won and "Win" or "Loss",

            }

            table.insert(tradeHistory, 1, entry)

            if #tradeHistory > 50 then

                table.remove(tradeHistory, #tradeHistory)

            end

        end

    end)

end

local function toggle(parent, label, key, onChange)

    local row = mk("Frame", {

        Size             = UDim2.new(1, -8, 0, 32),

        BackgroundColor3 = PAL.BG3,

        BorderSizePixel  = 0,

    }, parent)

    corner(6, row)

    mk("TextLabel", {

        Size             = UDim2.new(1, -50, 1, 0),

        Position         = UDim2.new(0, 10, 0, 0),

        BackgroundTransparency = 1,

        Text             = label,

        TextColor3       = PAL.TEXT,

        TextSize         = 13,

        Font             = Enum.Font.Gotham,

        TextXAlignment   = Enum.TextXAlignment.Left,

    }, row)

    local track = mk("Frame", {

        Size             = UDim2.new(0, 36, 0, 18),

        Position         = UDim2.new(1, -46, 0.5, -9),

        BackgroundColor3 = cfg[key] and PAL.ACCENT or PAL.BG,

        BorderSizePixel  = 0,

    }, row)

    corner(9, track)

    stroke(PAL.BORDER, 1, track)

    local thumb = mk("Frame", {

        Size             = UDim2.new(0, 12, 0, 12),

        Position         = cfg[key] and UDim2.new(1, -15, 0.5, -6) or UDim2.new(0, 3, 0.5, -6),

        BackgroundColor3 = PAL.WHITE,

        BorderSizePixel  = 0,

    }, track)

    corner(6, thumb)

    local tweenInfo = TweenInfo.new(0.15, Enum.EasingStyle.Quad)

    local function refresh()

        local on = cfg[key]

        TweenService:Create(track, tweenInfo, {

            BackgroundColor3 = on and PAL.ACCENT or PAL.BG,

        }):Play()

        TweenService:Create(thumb, tweenInfo, {

            Position = on

                and UDim2.new(1, -15, 0.5, -6)

                or  UDim2.new(0, 3,   0.5, -6),

        }):Play()

    end

    local btn = mk("TextButton", {

        Size             = UDim2.new(1, 0, 1, 0),

        BackgroundTransparency = 1,

        Text             = "",

    }, row)

    btn.MouseButton1Click:Connect(function()

        cfg[key] = not cfg[key]

        refresh()

        if onChange then onChange(cfg[key]) end

        saveConfig()

    end)

    return row

end

local function slider(parent, label, key, min, max, step, suffix)

    local row = mk("Frame", {

        Size             = UDim2.new(1, -8, 0, 52),

        BackgroundColor3 = PAL.BG3,

        BorderSizePixel  = 0,

    }, parent)

    corner(6, row)

        Position         = UDim2.new(0, 10, 0, 4),

        BackgroundTransparency = 1,

        Text             = label,

        TextColor3       = PAL.TEXT,

        TextSize         = 13,

        Font             = Enum.Font.Gotham,

        TextXAlignment   = Enum.TextXAlignment.Left,

    }, row)

    local valLbl = mk("TextLabel", {

        Size             = UDim2.new(0, 70, 0, 24),

        Position         = UDim2.new(1, -80, 0, 4),

        BackgroundTransparency = 1,

        Text             = tostring(cfg[key]) .. (suffix or ""),

        TextColor3       = PAL.ACCENT2,

        TextSize         = 13,

        Font             = Enum.Font.GothamBold,

        TextXAlignment   = Enum.TextXAlignment.Right,

    }, row)

    local track = mk("Frame", {

        Size             = UDim2.new(1, -20, 0, 4),

        Position         = UDim2.new(0, 10, 0, 36),

        BackgroundColor3 = PAL.BG,

        BorderSizePixel  = 0,

    }, row)

    corner(2, track)

    stroke(PAL.BORDER, 1, track)

    local pct = math.clamp((cfg[key] - min) / (max - min), 0, 1)

    local fill = mk("Frame", {

        Size             = UDim2.new(pct, 0, 1, 0),

        BackgroundColor3 = PAL.ACCENT,

        BorderSizePixel  = 0,

    }, track)

    corner(2, fill)

    local knob = mk("Frame", {

        Size             = UDim2.new(0, 12, 0, 12),

        Position         = UDim2.new(pct, -6, 0.5, -6),

        BackgroundColor3 = PAL.WHITE,

        BorderSizePixel  = 0,

    }, track)

    corner(6, knob)

    local dragging = false

    local btn = mk("TextButton", {

        Size             = UDim2.new(1, 0, 0, 30),

        Position         = UDim2.new(0, 0, 0, 22),

        BackgroundTransparency = 1,

        Text             = "",

    }, row)

    local function updateSlider(inputX)

        local abs     = track.AbsolutePosition.X

        local w       = track.AbsoluteSize.X

        local p       = math.clamp((inputX - abs) / w, 0, 1)

        local raw     = min + (max - min) * p

        local stepped = step and (math.round(raw / step) * step) or raw

        local val     = math.clamp(stepped, min, max)

        cfg[key] = val

        fill.Size     = UDim2.new(p, 0, 1, 0)

        knob.Position = UDim2.new(p, -6, 0.5, -6)

        valLbl.Text   = (type(val) == "number" and math.floor(val) == val)

            and tostring(val) .. (suffix or "")

            or  string.format("%.2f", val) .. (suffix or "")

        saveConfig()

    end

    btn.InputBegan:Connect(function(inp)

        if inp.UserInputType == Enum.UserInputType.MouseButton1 then

            dragging = true

            updateSlider(inp.Position.X)

        end

    end)

    btn.InputEnded:Connect(function(inp)

        if inp.UserInputType == Enum.UserInputType.MouseButton1 then

            dragging = false

        end

    end)

    UserInputService.InputChanged:Connect(function(inp)

        if dragging and inp.UserInputType == Enum.UserInputType.MouseMovement then

            updateSlider(inp.Position.X)

        end

    end)

    return row

end

local notifCount = 0

local function notify(text, color, duration)

    notifCount = notifCount + 1

    local card = mk("Frame", {

        Size             = UDim2.new(1, 0, 0, 36),

        BackgroundColor3 = PAL.BG2,

        BorderSizePixel  = 0,

        LayoutOrder      = -notifCount,

        BackgroundTransparency = 1,

    }, notifFrame)

    corner(6, card)

    stroke(color or PAL.ACCENT, 1, card)

    mk("TextLabel", {

        Size             = UDim2.new(1, -12, 1, 0),

        Position         = UDim2.new(0, 12, 0, 0),

        BackgroundTransparency = 1,

        Text             = text,

        TextColor3       = color or PAL.TEXT,

        TextSize         = 12,

        Font             = Enum.Font.Gotham,

        TextXAlignment   = Enum.TextXAlignment.Left,

        TextWrapped      = false,

    }, card)

    TweenService:Create(card, TweenInfo.new(0.2), {

        BackgroundTransparency = 0,

    }):Play()

    task.delay(duration or 3, function()

        TweenService:Create(card, TweenInfo.new(0.3), {

            BackgroundTransparency = 1,

            Size = UDim2.new(1, 0, 0, 0),

        }):Play()

        task.delay(0.35, function()

            card:Destroy()

        end)

    end)

end
